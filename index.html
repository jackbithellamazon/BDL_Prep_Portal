Database Migration — Required Columns
One-time setup — run both lines in your Supabase SQL editor. ship_segments lets a single SKU split across multiple shipments (8→FBA1, 2→FBA2) without merging. archived powers Archive Sent: archiving now hides rows from the prep sheet but keeps them on the Shipments page and in your stats. vat stores each product's VAT status (0% / 5% / 20%) for VAT-free flagging. The app works without these, but archived/split/VAT state won't survive a refresh until the columns exist. 
ALTER TABLE prep_rows ADD COLUMN IF NOT EXISTS ship_segments text DEFAULT '[]';
ALTER TABLE prep_rows ADD COLUMN IF NOT EXISTS archived boolean DEFAULT false;
ALTER TABLE prep_rows ADD COLUMN IF NOT EXISTS sheet_sync_id text;
ALTER TABLE prep_rows ADD COLUMN IF NOT EXISTS vat text DEFAULT '';
ALTER TABLE lavarion_asins ADD COLUMN IF NOT EXISTS vat text DEFAULT '20';
ALTER TABLE actions ADD COLUMN IF NOT EXISTS va_note text DEFAULT '';
ALTER TABLE actions ADD COLUMN IF NOT EXISTS jack_note text DEFAULT '';
ALTER TABLE actions ADD COLUMN IF NOT EXISTS reimbursed text DEFAULT 'No';
ALTER TABLE actions ADD COLUMN IF NOT EXISTS reimburse_note text DEFAULT '';
ALTER TABLE actions ADD COLUMN IF NOT EXISTS archived boolean DEFAULT false;
ALTER TABLE prep_rows ADD COLUMN IF NOT EXISTS chase_snooze_until timestamptz;
CREATE UNIQUE INDEX IF NOT EXISTS prep_rows_sheet_sync_id_key ON prep_rows (sheet_sync_id) WHERE sheet_sync_id IS NOT NULL;
CREATE TABLE IF NOT EXISTS app_settings (key text primary key, value text);
ALTER TABLE app_settings ENABLE ROW LEVEL SECURITY;
CREATE POLICY IF NOT EXISTS "app_settings_all" ON app_settings FOR ALL USING (true) WITH CHECK (true);
